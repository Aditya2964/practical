Prime number in class (C++)


// C++ program to print prime numbers in a range using class
#include <iostream>
using namespace std;

class prime
{
    private:
    int start, end;
    public:
    void getrange()
    {
        cout << "enter range: ";
        cin >> start >> end;
    }

    void isprime()
    {
        int i, j;
        cout << "prime numbers in the range " << start << " to " << end << " are: " << endl;
        for (i = start; i <= end; i++)
        {
            int check = 0;
            for (j = 2; j < i; j++)
            {
                if (i % j == 0)
                {
                    check++;
                    break;
                }
            }
            if (check == 0)
            {
                cout << i << " ";
            }
        }
        cout << endl;
    }
};

int main()
{
    prime p;
    p.getrange();
    p.isprime();
    return 0;
}

// C++ program to print palindrome numbers in a range using class
#include <iostream.h>
#include <conio.h>
class Palindrome
{
    private:
    int start, end;
    public:
    void getRange()
    {
        cout << "Enter range: ";
        cin >> start >> end;
    }
    void isPalindrome()
    {
        cout << "Palindrome numbers in the range " << start << " to " << end << " are: " << endl;
        for (int i = start; i <= end; i++)
        {
            int remain, result = 0;
            int number = i;
            while (number)
            {
                remain = number % 10;
                result = result * 10 + remain;
                number /= 10;
            }
            if (result == i)
            {
                cout << i << " ";
            }
        }
    }
};

int main()
{
    clrscr();
    Palindrome p;
    p.getRange();
    p.isPalindrome();
    getch();
}

        // C++ program to find given number is perfect or not using class
#include <iostream.h>
#include <conio.h>

class Perfect
{
    public:
    int number;
    void getNumber()
    {
        cout << "Enter a number: ";
        cin >> number;
    }
    void isPerfect()
    {
        int sum = 0;
        for (int i = 1; i < number; i++)
        {
            if (number % i == 0)
            {
                sum = sum + i;
            }
        }
        if (number == sum)
        {
            cout << number << " is a perfect number." << endl;
        }
        else
        {
            cout << number << " is not a perfect number." << endl;
        }
    }
};

int main()
{
    clrscr();
    Perfect p;
    p.getNumber();
    p.isPerfect();
    getch();
}

// C++ program to find factorial of a number using class
#include <iostream.h>
#include <conio.h>

class Factorial
{
    private:
    int number;
    public:
    void getNumber()
    {
        cout << "Enter a number: ";
        cin >> number;
    }
    void factorial()
    {
        int fact = 1;
        for (int i = 1; i <= number; i++)
        {
            fact *= i;
        }
        cout << "Factorial of " << number << " is: " << fact << endl;
    }
};

{
    clrscr();
    Factorial f;
    f.getNumber();
    f.factorial();

Operator Overloading


// C++ program to demonstrate operator overloading
#include <iostream.h>
#include <conio.h>
class Counter
{
    private:
    int value;
    public:
    Counter(int val = 0) : value(val) {}
    Counter operator++()
    {
        ++value;
        return *this;
    }
    Counter operator++(int)
    {
        Counter temp = *this;
        ++value;
        return temp;
    }
    void display() const
    {
        cout << "Counter value: " << value << endl;
    }
};
    int main()
    {
        clrscr();
        Counter c1(5);
        c1.display();
        ++c1;
        cout << "After pre-increment: ";
        c1.display();
        c1++;
        cout << "After post-increment: ";
        c1.display();
        getch();
    }

Linux commands


# Linux commands and attributes
    pwd : Prints the current working directory.
    Example : pwd

    ls : Lists files and directories in the current directory.
    Example : ls -l

    cd : Changes the current directory.
    Example : cd /home/user

    mkdir : Creates a new directory.
    Example : mkdir new-directory

    rmdir : Removes an empty directory.
    Example : rmdir old-directory

    rm : Removes files or directories.
    Example : rm file.txt

    cp : Copies files or directories.
    Example : cp source.txt destination.txt

    mv : Moves or renames files or directories.
    Example : mv oldname.txt newname.txt

    echo : Displays a line of text or a variable value.
    Example : echo "Hello, World!"

    more : Displays the contents of a file one screen at a time.
    Example : more file.txt

    less : Similar to more, but allows backward movement in the file.
    Example : less file.txt

    cal : Displays a calendar.
    Example : cal 2023

    date : Displays or sets the system date and time.
    Example : date / date +%d /date +%m / date +%Y

    who : Displays who is logged in.
    Example : who

    ps : Displays information about running processes.
    Example : ps aux

    kill : Terminates a process by its PID.
    Example : kill 1234

    wc : Counts lines, words, and characters in a file.
    Example : wc file.txt

    who am i : Displays the current user.
    Example : who am i
